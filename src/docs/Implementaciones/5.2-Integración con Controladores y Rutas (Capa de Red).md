## 🛠️ Funcionalidades del Endpoint `POST /api/pedidos`

### 1. Extracción y Validación de Entrada
El controlador extrae `usuarioId` y `productosComprados` directamente del cuerpo de la petición en formato JSON:
```json
{
  "usuarioId": 1,
  "productosComprados": [
    { "productoId": 1, "cantidad": 2 },
    { "productoId": 2, "cantidad": 1 }
  ]
}
```

### 2. Respuestas y Códigos de Estado HTTP

| Situación | Código HTTP | Formato de Respuesta |
| :--- | :--- | :--- |
| **Información Incompleta** (falta `usuarioId` o `productosComprados`, arrays vacíos, IDs no numéricos o cantidades <= 0) | `400 Bad Request` | `{"estado": "error", "mensaje": "Información incompleta..."}` |
| **Falta de Stock** (cualquier ítem supera el stock disponible en inventario) | `400 Bad Request` | `{"estado": "error", "mensaje": "Stock insuficiente para el producto..."}` |
| **Éxito en el Checkout** (pedido confirmado, stock descontado y detalles creados en PostgreSQL) | `201 Created` | `{"estado": "ok", "mensaje": "Pedido procesado y creado exitosamente", "datos": { ... }}` |

public crear = async (req: Request, res: Response, next: NextFunction): Promise<void> => {
  try {
    const { usuarioId, productosComprados, items } = req.body ?? {};
    const listaProductos = productosComprados ?? items;

    // Validación de usuarioId
    if (!usuarioId || typeof usuarioId !== "number" || usuarioId <= 0) {
      res.status(400).json({ estado: "error", mensaje: "..." });
      return;
    }

    // Validación de productosComprados
    if (!listaProductos || !Array.isArray(listaProductos) || listaProductos.length === 0) {
      res.status(400).json({ estado: "error", mensaje: "..." });
      return;
    }

    // Ejecución del checkout transaccional
    const pedidoFinalizado = await this.pedidoService.procesarCheckout({
      usuarioId,
      items: itemsNormalizados
    });

    res.status(201).json({
      estado: "ok",
      mensaje: "Pedido procesado y creado exitosamente",
      datos: pedidoFinalizado
    });
  } catch (error) {
    if (error instanceof InsufficientStockError || error instanceof BadRequestError) {
      res.status(400).json({ estado: "error", mensaje: error.message });
      return;
    }
    next(error);
  }
};

export function createPedidoRouter(customService?: PedidoService): Router {
  const router = Router();
  const service = customService ?? new PedidoService();
  const controller = new PedidoController(service);

  router.post("/", controller.crear); // POST /api/pedidos
  router.post("/checkout", controller.crear);
  router.get("/", controller.obtenerTodos);
  router.get("/:id", controller.obtenerPorId);

  return router;
}
